cmake_minimum_required(VERSION 3.24)
project(RTXVideoHDR LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows with D3D11 is required")
endif()
add_executable(RTXVideoHDRTest src/main.cpp src/pipeline.cpp)
target_compile_features(RTXVideoHDRTest PRIVATE cxx_std_20)
target_compile_definitions(RTXVideoHDRTest PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(RTXVideoHDRTest PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_link_libraries(RTXVideoHDRTest PRIVATE d3d11 dxgi d3dcompiler)
add_executable(RTXVideoHDRConvert src/convert_main.cpp src/pipeline.cpp src/child_process.cpp)
target_compile_features(RTXVideoHDRConvert PRIVATE cxx_std_20)
target_compile_definitions(RTXVideoHDRConvert PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(RTXVideoHDRConvert PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_link_libraries(RTXVideoHDRConvert PRIVATE d3d11 dxgi comdlg32 d3dcompiler)
set_property(TARGET RTXVideoHDRConvert RTXVideoHDRTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_executable(FilePathTests tests/file_paths.cpp src/pipeline.cpp)
target_include_directories(FilePathTests PRIVATE src)
target_compile_features(FilePathTests PRIVATE cxx_std_20)
target_compile_definitions(FilePathTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(FilePathTests PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_link_libraries(FilePathTests PRIVATE d3d11 dxgi d3dcompiler)
set_property(TARGET FilePathTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_test(NAME file_paths COMMAND FilePathTests)
add_executable(GpuColorTests tests/gpu_color.cpp src/pipeline.cpp)
target_include_directories(GpuColorTests PRIVATE src)
target_compile_features(GpuColorTests PRIVATE cxx_std_20)
target_compile_definitions(GpuColorTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(GpuColorTests PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_link_libraries(GpuColorTests PRIVATE d3d11 dxgi d3dcompiler)
set_property(TARGET GpuColorTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
