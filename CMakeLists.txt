cmake_minimum_required(VERSION 3.24)
project(RTXVideoHDR LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows with D3D11 is required")
endif()
add_executable(RTXVideoHDRTest src/main.cpp src/pipeline.cpp)
target_compile_features(RTXVideoHDRTest PRIVATE cxx_std_20)
target_compile_definitions(RTXVideoHDRTest PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(RTXVideoHDRTest PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_link_libraries(RTXVideoHDRTest PRIVATE d3d11 dxgi d3dcompiler)
add_executable(RTXVideoHDRConvert src/convert_main.cpp src/pipeline.cpp src/child_process.cpp)
